import numpy as np
import random

class EvolutionaryNeuralNetwork:
    def __init__(self, input_size, hidden_size, output_size):
        # Initialize random weights for input, hidden, and output layers
        self.input_size = input_size
        self.hidden_size = hidden_size
        self.output_size = output_size
        self.weights_input_hidden = np.random.randn(input_size, hidden_size)
        self.weights_hidden_output = np.random.randn(hidden_size, output_size)

    def forward(self, inputs):
        # Feedforward computation
        self.hidden_layer = np.tanh(np.dot(inputs, self.weights_input_hidden))
        self.output_layer = np.tanh(np.dot(self.hidden_layer, self.weights_hidden_output))
        return self.output_layer

    def mutate(self, mutation_rate=0.1):
        # Small random changes to weights to simulate evolution
        self.weights_input_hidden += mutation_rate * np.random.randn(*self.weights_input_hidden.shape)
        self.weights_hidden_output += mutation_rate * np.random.randn(*self.weights_hidden_output.shape)

    @staticmethod
    def crossover(parent1, parent2):
        # Creates a new network by blending parent weights
        child = EvolutionaryNeuralNetwork(parent1.input_size, parent1.hidden_size, parent1.output_size)
        child.weights_input_hidden = (parent1.weights_input_hidden + parent2.weights_input_hidden) / 2
        child.weights_hidden_output = (parent1.weights_hidden_output + parent2.weights_hidden_output) / 2
        return child

🌀 The Core Types of Cryptography
🔹 Symmetric Cryptography → Uses a single key to encrypt and decrypt data.

Example: AES (Advanced Encryption Standard)
🔹 Asymmetric Cryptography → Uses a public and private key pair for encryption/decryption.
Example: RSA (Rivest-Shamir-Adleman)
🔹 Hash Functions → One-way encryption that converts data into a fixed-length hash (used for password security).
Example: SHA-256 (used in Bitcoin and blockchain security)

from Crypto.Cipher import AES
import base64
import os

# Function to pad the message to match AES block size
def pad(s):
    return s + (AES.block_size - len(s) % AES.block_size) * chr(AES.block_size - len(s) % AES.block_size)

# Encrypt a message using AES
def encrypt_message(secret_key, message):
    secret_key = secret_key.ljust(32)[:32]  # Ensure key is 32 bytes long
    cipher = AES.new(secret_key.encode('utf-8'), AES.MODE_ECB)
    encrypted_text = cipher.encrypt(pad(message).encode('utf-8'))
    return base64.b64encode(encrypted_text).decode('utf-8')

# Decrypt a message
def decrypt_message(secret_key, encrypted_message):
    secret_key = secret_key.ljust(32)[:32]  # Ensure key is 32 bytes long
    cipher = AES.new(secret_key.encode('utf-8'), AES.MODE_ECB)
    decrypted_text = cipher.decrypt(base64.b64decode(encrypted_message)).decode('utf-8')
    return decrypted_text.rstrip()

# Example Usage
secret_key = "MySecretKey12345"  # Change this for security!
message = "Hello, cryptography!"

# Encrypt and decrypt the message
encrypted = encrypt_message(secret_key, message)
decrypted = decrypt_message(secret_key, encrypted)

print("Original Message:", message)
print("Encrypted Message:", encrypted)
print("Decrypted Message:", decrypted)

import hashlib
import time

# Simulating Bitcoin mining by solving a cryptographic puzzle
def mine_block(previous_hash, difficulty=4):
    nonce = 0
    start_time = time.time()
    
    while True:
        block_data = f"{previous_hash}{nonce}".encode()
        block_hash = hashlib.sha256(block_data).hexdigest()
        
        # Difficulty - The hash must start with '0000' (adjust difficulty as needed)
        if block_hash[:difficulty] == "0" * difficulty:
            end_time = time.time()
            print(f"Block mined! Hash: {block_hash}")
            print(f"Nonce: {nonce} | Time taken: {end_time - start_time:.2f} sec")
            return block_hash  # Return the mined block hash
        
        nonce += 1

# Example Usage: Mining a new block
